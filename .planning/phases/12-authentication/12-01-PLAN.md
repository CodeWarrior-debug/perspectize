       func (r *GormUserRepository) GetByClerkID(ctx context.Context, clerkID string) (*domain.User, error) {
           var gormUser GormUser
           result := r.db.WithContext(ctx).Where("clerk_user_id = ?", clerkID).First(&gormUser)
           if result.Error != nil {
               if errors.Is(result.Error, gorm.ErrRecordNotFound) {
                   return nil, fmt.Errorf("user not found for clerk ID: %s", clerkID)
               }
               return nil, result.Error
           }
           user := gormUserToDomain(&gormUser)
           return &user, nil
       }

    2. Create adapters/auth/clerk_middleware.go:
       - Middleware(userRepo UserRepository) func(http.Handler) http.Handler
       - Two layers:
         a. Clerk SDK's WithHeaderAuthorization() verifies JWT
         b. Custom handler extracts claims.Subject → calls userRepo.GetByClerkID
       - If no valid session → pass through as unauthenticated (permissive)
       - If valid session but no local user → log warning, pass as unauthenticated
         (on-demand creation deferred to Plan 03 webhook handler)
       - If valid session and local user found → inject into context

    See 12-RESEARCH-go-backend.md Pattern 1 for implementation reference.
