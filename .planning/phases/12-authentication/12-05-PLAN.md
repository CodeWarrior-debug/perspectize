       ```go
       // Example migration (run once):
       import clerkuser "github.com/clerk/clerk-sdk-go/v2/user"

       // For each real user (skip sentinel users):
       newClerkUser, err := clerkuser.Create(ctx, &clerkuser.CreateParams{
       ```
