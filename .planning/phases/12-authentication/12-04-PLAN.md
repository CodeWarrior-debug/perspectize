  <action>
    Create directives/auth.go:

    ```go
    package directives

    import (
        "context"
        "fmt"

        "github.com/99designs/gqlgen/graphql"
        "github.com/CodeWarrior-debug/perspectize/backend/internal/adapters/auth"
    )

    type DirectiveRoot struct{}

    func NewDirectiveRoot() *DirectiveRoot {
        return &DirectiveRoot{}
    }

    func (d *DirectiveRoot) Auth(ctx context.Context, obj interface{}, next graphql.Resolver) (interface{}, error) {
        user := auth.ForContext(ctx)
        if user == nil {
            return nil, fmt.Errorf("access denied: authentication required")
        }
        return next(ctx)
    }
    ```

    Wire in main.go via gqlgen's DirectiveRoot config.

    3. User resolvers:

       - Email visibility: In user query resolver, only return email if
         the requesting user's ID matches the queried user's ID

    Pattern for ownership check:
    ```go
    func (r *mutationResolver) DeletePerspective(ctx context.Context, id int) (bool, error) {
        user := auth.ForContext(ctx)
        // @auth directive already verified user != nil

        perspective, err := r.perspectiveService.GetByID(ctx, id)
        if err != nil {
            return false, err
        }
        if perspective.UserID != user.ID {
            return false, fmt.Errorf("access denied: not the owner of this perspective")
        }
        return r.perspectiveService.Delete(ctx, id)
    }
    ```

    Pattern for email visibility:
    ```go
    func (r *userResolver) Email(ctx context.Context, obj *model.User) (*string, error) {
        user := auth.ForContext(ctx)
        if user == nil || user.ID != obj.ID {
            return nil, nil  // Hide email for unauthenticated or other users
        }
        return &obj.Email, nil
    }
    ```
