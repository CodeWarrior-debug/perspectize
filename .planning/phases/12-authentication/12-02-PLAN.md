<task type="auto">
  <name>Update GraphQL client for Bearer token forwarding</name>
  <files>
    frontend/src/lib/queries/client.ts
  </files>
  <action>
    Update the GraphQL client to include Bearer token from Clerk:

    ```typescript
    import { GraphQLClient } from 'graphql-request';

    const GRAPHQL_ENDPOINT = import.meta.env.VITE_GRAPHQL_URL || 'http://localhost:8080/graphql';

    export const graphqlClient = new GraphQLClient(GRAPHQL_ENDPOINT);

    /**
     * Get current Clerk session token for API requests.
     * Returns null if not authenticated.
     * Token is cached by ClerkJS with ~1 min TTL.
     */
    export async function getAuthToken(): Promise<string | null> {
      try {
        const token = await window.Clerk?.session?.getToken();
        return token ?? null;
      } catch {
        return null;
      }
    }

    /**
     * Make a GraphQL request with optional auth.
     * Automatically includes Bearer token if user is signed in.
     */
    export async function graphqlRequest<T>(
      document: string,
      variables?: Record<string, unknown>
    ): Promise<T> {
      const token = await getAuthToken();
      const headers: Record<string, string> = {};
      if (token) {
        headers['Authorization'] = `Bearer ${token}`;
      }
      return graphqlClient.request<T>(document, variables, headers);
    }
    ```

    Update all existing query/mutation callers to use graphqlRequest() instead of
    graphqlClient.request() directly. This ensures auth tokens are always included.
  </action>

  </verify>
