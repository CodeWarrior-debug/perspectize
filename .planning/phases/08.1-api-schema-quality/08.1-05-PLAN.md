// User resolves the user field for a perspective
func (r *perspectiveResolver) User(ctx context.Context, obj *model.Perspective) (*model.User, error) {
    // obj.UserID is string (GraphQL ID type), convert to int
    userID, err := strconv.Atoi(obj.UserID)
    if err != nil {
        return nil, fmt.Errorf("invalid user ID: %w", err)
    }

    // Fetch user via service
    user, err := r.UserService.GetByID(ctx, userID)
    if err != nil {
        if errors.Is(err, domain.ErrNotFound) {
            // User deleted after perspective created - return nil (allowed by schema User nullable)
            return nil, nil
        }
        return nil, fmt.Errorf("failed to fetch user: %w", err)
    }

    // Convert domain user to GraphQL model
    return userDomainToModel(user), nil
}
```

**Key decisions:**
- UserID is string in GraphQL model (ID scalar) - convert to int for service call
- Return nil on ErrNotFound (schema allows null User) - graceful degradation if user deleted
- Use existing userDomainToModel converter (should already exist in resolvers)

**Import additions** (add at top if not present):

// Content resolves the content field for a perspective
func (r *perspectiveResolver) Content(ctx context.Context, obj *model.Perspective) (*model.Content, error) {
    // obj.ContentID is *string (nullable GraphQL ID), check for nil
    if obj.ContentID == nil {
        // Perspective not linked to content yet - return nil (allowed by schema)
        return nil, nil
    }

    // Convert string ID to int
    contentID, err := strconv.Atoi(*obj.ContentID)
    if err != nil {
        return nil, fmt.Errorf("invalid content ID: %w", err)
    }

    // Fetch content via service
    content, err := r.ContentService.GetByID(ctx, contentID)
    if err != nil {
        if errors.Is(err, domain.ErrNotFound) {
            // Content deleted after perspective created - return nil
            return nil, nil
        }
        return nil, fmt.Errorf("failed to fetch content: %w", err)
    }

    // Convert domain content to GraphQL model
    return domainToModel(content), nil
}
```

**Key decisions:**
- ContentID is *string (nullable pointer) - check for nil before conversion
- Return nil on ErrNotFound (schema allows null Content) - graceful degradation
- Use existing domainToModel converter for Content (should exist in content_resolver.go)
