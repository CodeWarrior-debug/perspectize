```typescript
import type { HandleClientError } from '@sveltejs/kit';
import { toast } from 'svelte-sonner';

export const handleError: HandleClientError = ({ error, event, status, message }) => {
  // Log to console for debugging
  console.error('Client error caught by handleError hook:', {
    error,
    event,
    status,
    message,
    url: event.url.href
  });

  // Show user-friendly toast for network errors
  if (error instanceof TypeError && error.message.includes('fetch')) {
    toast.error('Network error. Please check your connection and try again.');
  }

  // Return sanitized error message (exposed to user via $page.error)
  return {
    message: message || 'An unexpected error occurred',
    code: status || 500
  };
};
```

IMPORTANT: This hook only runs for errors in **load functions** (+page.ts, +layout.ts), NOT for errors inside component code. Component errors are caught by +error.svelte boundary.

The hook serves two purposes:
1. Log errors for debugging (console.error)
2. Show user-friendly notifications for specific error types (network errors get toast)
3. Sanitize error messages before exposing to user via $page.error

Do NOT rely on this for all error handling — it's a last-resort catch for load function failures.

```typescript
import { GraphQLClient } from 'graphql-request';

// Auto-upgrade HTTP to HTTPS in production (M-25)
let endpoint = import.meta.env.VITE_GRAPHQL_URL || 'http://localhost:8080/graphql';
if (import.meta.env.PROD && endpoint.startsWith('http://')) {
  endpoint = endpoint.replace('http://', 'https://');
  console.warn('[GraphQL Client] Auto-upgraded endpoint to HTTPS for production:', endpoint);
}

const GRAPHQL_ENDPOINT = endpoint;
const REQUEST_TIMEOUT = 10000; // 10 seconds

// Create AbortController for timeout
function createAbortController(): AbortController {
  const controller = new AbortController();
  setTimeout(() => controller.abort(), REQUEST_TIMEOUT);
  return controller;
}

export const graphqlClient = new GraphQLClient(GRAPHQL_ENDPOINT, {
  headers: {
    // Future: Add auth header when authentication is implemented
    // 'Authorization': `Bearer ${getToken()}`
  },
  fetch: (url, options) => {
    const controller = createAbortController();

    return fetch(url, {
      ...options,
      signal: controller.signal
    }).then(async (response) => {
      // Error interceptor: check for non-2xx responses
      if (!response.ok) {
        const text = await response.text();
        const error = new Error(`GraphQL request failed: ${response.status}`);
        (error as any).status = response.status; // Attach status for retry logic
        (error as any).response = { status: response.status, body: text };
        throw error;
      }
      return response;
    }).catch((error) => {
      // Handle timeout specifically
      if (error.name === 'AbortError') {
        throw new Error('GraphQL request timed out after 10 seconds');
      }
      throw error;
    });
  }
});
```

This adds:
1. **Timeout:** 10-second AbortController timeout on all requests
2. **HTTPS upgrade:** Automatically converts http:// to https:// in production (fixes M-25)
3. **Error interceptor:** Attaches status code to errors for TanStack Query retry logic
4. **Timeout error message:** User-friendly message when timeout occurs

graphql-request doesn't have built-in timeout/interceptor APIs, so we override the `fetch` option with custom logic.

REPLACE with selective retry configuration:
```typescript
const queryClient = new QueryClient({
  defaultOptions: {
    queries: {
      enabled: browser,
      retry: (failureCount, error) => {
        // Max 3 retry attempts
        if (failureCount >= 3) return false;

        // Retry network errors (fetch failures, timeouts)
        if (error instanceof TypeError) return true;

        // Retry 5xx server errors
        if ('status' in error && typeof error.status === 'number') {
          return error.status >= 500 && error.status < 600;
        }

        // Don't retry 4xx client errors (validation, not found, etc.)
        return false;
      },
      retryDelay: (attemptIndex) => {
        // Exponential backoff: 1s, 2s, 4s, capped at 30s
        return Math.min(1000 * 2 ** attemptIndex, 30000);
      }
    }
  }
});
```

This implements M-26 (only retry 5xx/network errors). The retry function checks:
1. **failureCount >= 3**: Stop after 3 retries
2. **TypeError**: Network errors (fetch failures, CORS, timeout) — RETRY
3. **status >= 500**: Server errors — RETRY
4. **All others** (including 4xx validation errors): DO NOT RETRY

The retryDelay uses exponential backoff to avoid hammering the server.
