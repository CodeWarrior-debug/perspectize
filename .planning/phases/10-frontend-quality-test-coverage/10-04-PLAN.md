func TestIntID_MarshalGQL_EdgeCases(t *testing.T) {
	tests := []struct {
		name     string
		input    graphql.IntID
		wantJSON string
	}{
		{
			name:     "zero value",
			input:    graphql.IntID(0),
			wantJSON: "0",
		},
		{
			name:     "negative value",
			input:    graphql.IntID(-1),
			wantJSON: "-1",
		},
		{
			name:     "large positive value",
			input:    graphql.IntID(2147483647), // Max int32
			wantJSON: "2147483647",
		},
	}

	for _, tt := range tests {
		t.Run(tt.name, func(t *testing.T) {
			var buf bytes.Buffer
			writer := bufio.NewWriter(&buf)

			tt.input.MarshalGQL(writer)
			writer.Flush()

			assert.Equal(t, tt.wantJSON, buf.String())
		})
	}
}

func TestIntID_UnmarshalGQL_InvalidTypes(t *testing.T) {
	tests := []struct {
		name    string
		input   interface{}
		wantErr bool
	}{
		{
			name:    "string value rejected",
			input:   "123",
			wantErr: true,
		},
		{
			name:    "float value rejected",
			input:   123.45,
			wantErr: true,
		},
		{
			name:    "bool value rejected",
			input:   true,
			wantErr: true,
		},
		{
			name:    "nil value rejected",
			input:   nil,
			wantErr: true,
		},
		{
			name:    "valid int accepted",
			input:   123,
			wantErr: false,
		},
		{
			name:    "zero int accepted",
			input:   0,
			wantErr: false,
		},
	}

	for _, tt := range tests {
		t.Run(tt.name, func(t *testing.T) {
			var id graphql.IntID
			err := id.UnmarshalGQL(tt.input)

			if tt.wantErr {
				assert.Error(t, err)
			} else {
				assert.NoError(t, err)
				if intVal, ok := tt.input.(int); ok {
					assert.Equal(t, graphql.IntID(intVal), id)
				}
			}
		})
	}
}
