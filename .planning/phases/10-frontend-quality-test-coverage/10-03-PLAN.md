func TestResolver_Users(t *testing.T) {
	tests := []struct {
		name      string
		setupMock func(*mockUserService)
		wantCount int
		wantErr   bool
	}{
		{
			name: "success - returns multiple users",
			setupMock: func(ms *mockUserService) {

					return []*domain.User{

						{ID: 2, Username: "user2", Email: "user2@example.com"},
						{ID: 3, Username: "user3", Email: "user3@example.com"},
					}, nil
				}
			},
			wantCount: 3,
			wantErr:   false,
		},
		{
			name: "success - returns empty list when no users",
			setupMock: func(ms *mockUserService) {
				ms.listAllFn = func(ctx context.Context) ([]*domain.User, error) {
					return []*domain.User{}, nil
				}
			},
			wantCount: 0,
			wantErr:   false,
		},
		{
			name: "error - service returns error",
			setupMock: func(ms *mockUserService) {
				ms.listAllFn = func(ctx context.Context) ([]*domain.User, error) {
					return nil, domain.ErrInternal
				}
			},
			wantCount: 0,
			wantErr:   true,
		},
	}

	for _, tt := range tests {
		t.Run(tt.name, func(t *testing.T) {
			mockService := &mockUserService{}
			tt.setupMock(mockService)

			// Create resolver with mock services
			resolver := resolvers.NewResolver(nil, mockService, nil)

			if tt.wantErr {
				assert.Error(t, err)
				assert.Nil(t, users)
			} else {
				require.NoError(t, err)
				assert.NotNil(t, users)
				assert.Len(t, users, tt.wantCount)
			}
		})
	}
}
