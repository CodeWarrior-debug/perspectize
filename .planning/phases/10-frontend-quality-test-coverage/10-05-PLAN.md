package repositories_test

import (
	"context"
	"os"
	"testing"

	"github.com/CodeWarrior-debug/perspectize/backend/internal/adapters/repositories/postgres"
	"github.com/CodeWarrior-debug/perspectize/backend/internal/core/domain"
	"github.com/CodeWarrior-debug/perspectize/backend/pkg/database"
	"github.com/stretchr/testify/assert"
	"github.com/stretchr/testify/require"
	"gorm.io/gorm"
)

// setupTestDB creates a test database connection or skips test if DB unavailable
func setupTestDB(t *testing.T) *gorm.DB {
	t.Helper()

	databaseURL := os.Getenv("DATABASE_URL")
	if databaseURL == "" {
		t.Skip("Skipping integration test: DATABASE_URL not set")
	}

	db, err := database.ConnectGORM(databaseURL)
	if err != nil {
		t.Skipf("Skipping integration test: database unavailable: %v", err)
	}

	return db
}

// cleanupContent deletes all content records for test isolation
func cleanupContent(t *testing.T, db *gorm.DB) {
	t.Helper()
	// Delete in order: perspectives first (foreign key), then content
	db.Exec("DELETE FROM perspectives")
	db.Exec("DELETE FROM content")
}

func TestContentRepository_List_Pagination(t *testing.T) {
	db := setupTestDB(t)
	defer cleanupContent(t, db)

	repo := postgres.NewGormContentRepository(db)

	// Create 5 test items
	for i := 1; i <= 5; i++ {
		_, err := repo.Create(context.Background(), &domain.Content{
			Name:        "Test Item " + string(rune('A'+i-1)),
			URL:         "https://example.com/" + string(rune('a'+i-1)),
			ContentType: "YOUTUBE",
		})
		require.NoError(t, err)
	}

	// Test: Fetch first 2 items
	result, err := repo.List(context.Background(), domain.ContentListParams{
		First: intPtr(2),
		Sort: &domain.ContentSortInput{
			By:    domain.ContentSortByName,
			Order: domain.SortOrderAsc,
		},
	})

	require.NoError(t, err)
	assert.Len(t, result.Items, 2)
	assert.True(t, result.PageInfo.HasNextPage, "Should have more items")
	assert.NotNil(t, result.PageInfo.EndCursor, "Should have end cursor")

	// Test: Fetch next page using cursor
	result2, err := repo.List(context.Background(), domain.ContentListParams{
		First: intPtr(2),
		After: result.PageInfo.EndCursor,
		Sort: &domain.ContentSortInput{
			By:    domain.ContentSortByName,
			Order: domain.SortOrderAsc,
		},
	})

	require.NoError(t, err)
	assert.Len(t, result2.Items, 2)
	assert.NotEqual(t, result.Items[0].ID, result2.Items[0].ID, "Second page should have different items")
}

func TestPerspectiveRepository_Update(t *testing.T) {
	db := setupTestDB(t)
	defer cleanupPerspectives(t, db)

	userRepo := postgres.NewGormUserRepository(db)
	contentRepo := postgres.NewGormContentRepository(db)
	perspRepo := postgres.NewGormPerspectiveRepository(db)

	// Create test data
	user, err := userRepo.Create(context.Background(), &domain.User{
		Username: "testuser",
		Email:    "test@example.com",
	})
	require.NoError(t, err)

	content, err := contentRepo.Create(context.Background(), &domain.Content{
		Name:        "Test Content",
		URL:         "https://example.com/test",
		ContentType: "YOUTUBE",
	})
	require.NoError(t, err)

	// Create perspective
	created, err := perspRepo.Create(context.Background(), &domain.Perspective{
		UserID:    user.ID,
		ContentID: content.ID,
		Quality:   intPtr(3),
		Like:      stringPtr("Initial like"),
	})
	require.NoError(t, err)

	// Test: Update perspective
	created.Quality = intPtr(5)
	created.Like = stringPtr("Updated like")

	updated, err := perspRepo.Update(context.Background(), created)
	require.NoError(t, err)
	assert.Equal(t, 5, *updated.Quality)
	assert.Equal(t, "Updated like", *updated.Like)

	// Verify update persisted to DB
	fetched, err := perspRepo.GetByID(context.Background(), created.ID)
	require.NoError(t, err)
	assert.Equal(t, 5, *fetched.Quality)
	assert.Equal(t, "Updated like", *fetched.Like)
}
