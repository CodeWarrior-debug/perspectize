  <action>
Create `frontend/codegen.ts`:

```typescript
import type { CodegenConfig } from '@graphql-codegen/cli';

const config: CodegenConfig = {
  schema: '../backend/schema.graphql', // Use file path (not introspection) to avoid requiring running server during CI
  documents: ['src/lib/queries/**/*.ts'],
  ignoreNoDocuments: true, // Don't error if no query files found during initial run
  generates: {
    './src/lib/graphql/': {
      preset: 'client',
      config: {
        useTypeImports: true // Generate `import type` instead of `import` for type-only imports
      },
      plugins: []
    }
  }
};

export default config;
```

Update `frontend/package.json` scripts:
- Add `"codegen": "graphql-codegen --config codegen.ts"`
- Update `"dev"` to `"pnpm run codegen && vite dev"` (run codegen before starting dev server)
- Add `"prebuild"` script: `"pnpm run codegen"` (automatically runs before `pnpm run build`)

This ensures types are always fresh before builds. The schema path uses relative path to backend so codegen works offline (no running server required).

**frontend/src/lib/queries/content.ts:**

BEFORE (manual types):
```typescript
import { gql } from 'graphql-request';

export interface ContentItem { id: string; name: string; ... }
export interface ContentResponse { content: { items: ContentItem[]; ... } }

export const LIST_CONTENT = gql`...`;
```

AFTER (generated types):
```typescript
import { graphql } from '$lib/graphql'; // Use generated graphql() function
import type { ResultOf } from '$lib/graphql'; // For extracting result types

export const LIST_CONTENT = graphql(`
  query ListContent($first: Int, $after: String, $sort: ContentSortInput, $filter: ContentFilterInput) {
    content(first: $first, after: $after, sort: $sort, filter: $filter) {
      items {
        id
        name
        url
        contentType
        description
        channelTitle
        publishedAt
        tags
        viewCount
        likeCount
        duration
        createdAt
        updatedAt
      }
      pageInfo {
        hasNextPage
        hasPreviousPage
        startCursor
        endCursor
      }
      totalCount
    }
  }
`);

// Extract types from generated query
export type ContentListResult = ResultOf<typeof LIST_CONTENT>;
export type ContentItem = ContentListResult['content']['items'][number];
export type ContentResponse = ContentListResult;
```

DELETE all manual interface definitions (ContentItem, ContentResponse, etc.). Types are now inferred from the query via `ResultOf`.
